Мета роботи

Завдання 1. Підрахунок та сума парних елементів черги

Умова: Створити чергу з 15 цілих чисел. Підрахувати кількість та суму парних елементів черги.
Програмний код:

#include <iostream>
#include <queue>
using namespace std;

int main() {
    queue<int> numbers;

    // Додаємо 15 цілих чисел до черги
    numbers.push(4);
    numbers.push(7);
    numbers.push(10);
    numbers.push(3);
    numbers.push(8);
    numbers.push(15);
    numbers.push(22);
    numbers.push(33);
    numbers.push(40);
    numbers.push(55);
    numbers.push(60);
    numbers.push(77);
    numbers.push(80);
    numbers.push(91);
    numbers.push(100);

    int countEven = 0;
    int sumEven = 0;

    cout << "Початкова черга: ";

    // Витягуємо елементи, виводимо їх та робимо обчислення
    while (!numbers.empty()) {
        int x = numbers.front();
        cout << x << " "; // Виводимо елемент черги
        numbers.pop();

        // Перевірка на парність
        if (x % 2 == 0) {
            countEven++;
            sumEven += x;
        }
    }
    cout << endl;

    // Виведення результатів
    cout << "Кількість парних чисел: " << countEven << endl;
    cout << "Сума парних чисел: " << sumEven << endl;

    return 0;
}

Завдання 2. Фільтрація черги чисел

Умова: Додайте до черги цілі числа від 1 до 100. Створіть нову чергу, що містить лише числа, які діляться на 4 без остачі. Виведіть нову чергу.
Програмний код:

#include <iostream>
#include <queue>
using namespace std;

int main() {
    queue<int> source;
    queue<int> result;

    // Записуємо в чергу цілі числа від 1 до 100
    for (int i = 1; i <= 100; i++) {
        source.push(i);
    }

    // Витягуємо по одному і записуємо в нову чергу числа, що діляться на 4
    while (!source.empty()) {
        int x = source.front();
        source.pop();

        if (x % 4 == 0) {
            result.push(x);
        }
    }

    cout << "Числа, що діляться на 4: ";

    // Виводимо нову чергу
    while (!result.empty()) {
        cout << result.front() << " ";
        result.pop();
    }
    cout << endl;

    return 0;
}

Завдання 3. Черга символів та підрахунок голосних

Умова: Створіть чергу символів, що містить літери англійського слова. Підрахуйте кількість голосних літер у слові.
Програмний код:

#include <iostream>
#include <queue>
#include <string>
using namespace std;

int main() {
    queue<char> letters;
    string text = "education"; 
    
    // Записуємо кожну літеру слова в чергу
    for (int i = 0; i < text.length(); i++) {
        letters.push(text[i]);
    }
    
    int vowelCount = 0;
    
    // Витягуємо літери з черги та перевіряємо, чи є вони голосними
    while (!letters.empty()) {
        char c = letters.front();
        letters.pop();
        
        // Перевірка на голосні літери
        if (c == 'a' || c == 'e' || c == 'i' || c == 'o' || c == 'u' || c == 'y' ||
            c == 'A' || c == 'E' || c == 'I' || c == 'O' || c == 'U' || c == 'Y') {
            vowelCount++;
        }
    }
    
    cout << "Слово: '" << text << "'" << endl;
    cout << "Кількість голосних літер у слові: " << vowelCount << endl;
    
    return 0;
}

Завдання 4. Черга структур (Студенти)

Умова: Створити чергу, інформаційними полями якої є прізвище та середній бал студента. Додати у чергу відомості про студентів. Вивести список студентів, середній бал яких >= 80.
Програмний код:

#include <iostream>
#include <queue>
#include <string>
using namespace std;

struct Student {
    string surname;
    int mark;
};

int main() {
    queue<Student> students;

    students.push({"Ivanenko", 85});
    students.push({"Petrenko", 74});
    students.push({"Kovalenko", 92});
    students.push({"Melnyk", 78});

    cout << "Студенти з балом не менше 80:\n";

    while (!students.empty()) {
        Student s = students.front();
        students.pop();

        if (s.mark >= 80) {
            cout << s.surname
                 << " - " << s.mark << endl;
        }
    }

    return 0;
}
